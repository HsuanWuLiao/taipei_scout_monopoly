Google Sheet 雲端資料庫連線設定

未填寫時預設以「本機演示模式」運行，內建完整雙北 12 站與題庫，可立即遊玩。

Google Apps Script 橋接程式碼（貼入試算表「擴充功能 → Apps Script」） 
function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const read = (name) => {
    const sh = ss.getSheetByName(name);
    if (!sh) return [];
    const [h, ...rows] = sh.getDataRange().getValues();
    return rows.filter(r => r[0] !== '').map(r => Object.fromEntries(h.map((k, i) => [k, r[i]])));
  };
  return ContentService.createTextOutput(JSON.stringify({
    stations: read('Stations'), questions: read('Questions'),
    players: read('Players'), transactions: read('Transactions').slice(-20)
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const data = JSON.parse(e.postData.contents);
  if (data.action === 'log' && data.row) {
    ss.getSheetByName('Transactions').appendRow(data.row);
  }
  if (data.action === 'syncPlayers' && Array.isArray(data.players)) {
    const sh = ss.getSheetByName('Players');
    if (sh.getLastRow() > 1) sh.getRange(2, 1, sh.getLastRow() - 1, 8).clearContent();
    data.players.forEach((p, idx) => sh.getRange(idx + 2, 1, 1, 8).setValues([[
      p.id, p.name, p.color, p.pos, p.cash, p.propsCount, p.totalAsset, new Date().toLocaleString('zh-TW')
    ]]));
  }
  if (data.action === 'syncStations' && Array.isArray(data.stations)) {
    const sh = ss.getSheetByName('Stations');
    data.stations.forEach((s, idx) => {
      sh.getRange(idx + 2, 9, 1, 2).setValues([[s.purchaseCount, s.owners.join('、')]]);
    });
  }
  return ContentService.createTextOutput(JSON.stringify({ ok: true })).setMimeType(ContentService.MimeType.JSON);
}
